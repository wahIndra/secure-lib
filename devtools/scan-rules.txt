# secure-scan rules, shared by the Java and JavaScript scanners.
#
# One rule per block, blocks separated by a blank line. Keys:
#   id           unique rule id (shown in reports and used in suppressions)
#   lang         comma list of: java (also .kt/.groovy/.scala), js (also .ts/.jsx/.tsx), template, config
#   severity     HIGH | MEDIUM | LOW
#   owasp        OwaspCategory constant name
#   scope        code = string-literal contents are masked before matching (default)
#                raw  = match the line as written (needed when the literal itself matters)
#   flags        i = case-insensitive
#   pattern      regex, using only syntax common to Java and JavaScript (no inline (?i), no possessive;
#                escape "[" inside a character class, since Java treats [[ as a nested class)
#   unless       optional regex: skip the line if it matches (e.g. it already uses the safe API)
#   unless-file  optional regex: skip the whole file if it matches anywhere
#   message      what is wrong
#   fix          how to fix it, preferably with this library
#
# Suppress a reviewed finding with a comment on the same line or the line above:
#   // secure-scan:ignore SQL_STRING_CONCAT -- column list is a compile-time constant

# ------------------------------------------------------------------ A05 Injection: SQL

id: SQL_STRING_CONCAT
lang: java, js
severity: HIGH
owasp: INJECTION
scope: raw
flags: i
pattern: (?<!\\)"\s*(?:select\s[^"]*\sfrom\b|insert\s+into\b|update\s+\w+\s+set\b|delete\s+from\b|merge\s+into\b)[^"]*(?<!\\)"\s*\+\s*[\w(]|(?<!\\)'\s*(?:select\s[^']*\sfrom\b|insert\s+into\b|update\s+\w+\s+set\b|delete\s+from\b|merge\s+into\b)[^']*(?<!\\)'\s*\+\s*[\w(]
message: SQL statement built by string concatenation
fix: Use a PreparedStatement / parameterized query with ? (or $1) placeholders. Identifiers such as ORDER BY columns cannot be parameters: use SafeSql.allowlisted(...) or SafeSql.mapped(...).

id: SQL_CLAUSE_CONCAT
lang: java, js
severity: MEDIUM
owasp: INJECTION
scope: raw
flags: i
pattern: (?<!\\)["'][^"']*\b(?:where|order\s+by|group\s+by|having|values\s*\(|(?:select|from|where|by)\b[^"']*\b(?:limit|offset))\s*(?:[\w.]+\s*(?:=|<>|!=|<|>|like|in)\s*)?\(?'?(?<!\\)["']\s*\+\s*[A-Za-z_]
unless: ["']\s*(?:select\s.*\sfrom\b|insert\s+into\b|update\s+\w+\s+set\b|delete\s+from\b|merge\s+into\b)
message: SQL clause continued with a concatenated value
fix: Bind the value as a parameter. For sort columns use SafeSql.allowlisted(...) and SafeSql.direction(...).

id: SQL_TEMPLATE_LITERAL
lang: js
severity: HIGH
owasp: INJECTION
scope: raw
flags: i
pattern: (?<![\w.$])`\s*(?:select\s[^`]*\sfrom\b|insert\s+into\b|update\s+\w+\s+set\b|delete\s+from\b)[^`]*\$\{
message: SQL statement built with an untagged template literal
fix: Pass values as parameters (db.query('... WHERE id = $1', [id])) or use a tagged template that parameterizes (sql`...${id}`).

id: SQL_FORMAT
lang: java
severity: HIGH
owasp: INJECTION
scope: raw
flags: i
pattern: (?:String\.format|\.formatted)\s*\(\s*"\s*(?:select|insert|update|delete|merge)\b
message: SQL statement built with String.format
fix: Use a PreparedStatement with ? placeholders.

id: SQL_EXECUTE_CONCAT
lang: java, js
severity: HIGH
owasp: INJECTION
scope: raw
pattern: \.(?:executeQuery|executeUpdate|executeLargeUpdate|execute|prepareStatement|prepareCall|addBatch|createQuery|createNativeQuery|createSQLQuery|queryForList|queryForObject|queryForMap|queryForRowSet|query|raw)\s*\(\s*(?:[\w.]+\s*\+\s*[\w"']|"[^"]*"\s*\+|'[^']*'\s*\+)
message: Query executed with a concatenated string
fix: Keep the SQL text constant and bind values as parameters.

id: SQL_RAW_UNSAFE
lang: js
severity: HIGH
owasp: INJECTION
pattern: \$(?:queryRawUnsafe|executeRawUnsafe)\s*\(|\bsequelize\.literal\s*\([^)]*(?:req\.|\+|\$\{)|\.whereRaw\s*\([^,)]*(?:\+|\$\{)
message: Raw/unsafe ORM query with dynamic SQL
fix: Use the ORM's parameterized form ($queryRaw tagged template, whereRaw('x = ?', [v])).

id: NOSQL_INJECTION
lang: js
severity: HIGH
owasp: INJECTION
scope: raw
pattern: \.(?:find|findOne|findOneAndUpdate|findOneAndDelete|updateOne|updateMany|deleteOne|deleteMany|countDocuments|aggregate)\s*\(\s*req\.(?:body|query|params)\b|["']\$where["']
message: NoSQL query built directly from request data ($ne / $gt / $where operator injection)
fix: Build the filter from validated scalar fields, e.g. { email: String(req.body.email) }, and reject objects where strings are expected.

# ------------------------------------------------------------------ A01 Broken Access Control: path traversal, redirects, SSRF, CORS
